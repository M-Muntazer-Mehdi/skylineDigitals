# SMTP Configuration for Email Sending
# Copy this file to .env.local and fill in your SMTP details

# SMTP Server Settings
# For Gmail: smtp.gmail.com
# For Outlook: smtp-mail.outlook.com
# For Yahoo: smtp.mail.yahoo.com
# For custom SMTP: your-smtp-server.com
SMTP_HOST=smtp.gmail.com

# SMTP Port (usually 587 for TLS, 465 for SSL, 25 for non-encrypted)
SMTP_PORT=587

# SMTP Username (your email address)
SMTP_USER=your-email@gmail.com

# SMTP Password (for Gmail, use App Password, not regular password)
# Get App Password: https://myaccount.google.com/apppasswords
SMTP_PASSWORD=your-app-password-here

# Email Configuration
# The email address where contact form submissions will be sent
CONTACT_EMAIL=skylinedigitals.tech@gmail.com

# From Email (optional - defaults to SMTP_USER)
# The email address that will appear as the sender
FROM_EMAIL=skylinedigitals.tech@gmail.com
